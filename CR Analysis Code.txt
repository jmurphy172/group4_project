### Adding a column for rating

def rate_elo(elo):
    try:
        elo = int(elo)  
    except ValueError:
        return "Unrated" 
    
    if 1000 <= elo <= 1999:
        return "Low rated"
    elif elo >= 2000:
        return "High rated"
    else:
        return "Unrated"

no_new_comers["WhiteRating"] = no_new_comers["WhiteElo"].apply(rate_elo)
no_new_comers["BlackRating"] = no_new_comers["BlackElo"].apply(rate_elo)


### Adding a column to label openings as uncommon or common

opening_counts = no_new_comers["Opening"].value_counts()
n_total_openings = len(opening_counts)
n_uncommon = int(n_total_openings * 0.2)


uncommon_openings = opening_counts.tail(n_uncommon).index


no_new_comers['OpeningType'] = no_new_comers['Opening'].apply(
    lambda x: 'Uncommon' if x in uncommon_openings else 'Common'
)
